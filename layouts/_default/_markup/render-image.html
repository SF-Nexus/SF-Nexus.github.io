{{- /* Every markdown image is served as responsive webp. Charts and screenshots link to the full-size original so dense figures stay legible. Give an image a markdown title to turn it into a captioned figure: ![Alt text](/images/foo.png "Caption text") */ -}} {{- $img := partial "responsive-image.html" (dict "src" .Destination "page" .Page "alt" (.Text | plainify | htmlUnescape) "sizes" "(min-width: 64em) 46rem, 94vw") -}} {{- /* Site-rooted paths go through relURL; page-relative ones (a bundle's own images) pass through. */ -}} {{- $href := cond (strings.HasPrefix .Destination "/") (.Destination | relURL) .Destination -}} {{- $link := printf `%s` $href $img -}} {{- if .IsBlock -}} {{- else -}} {{ $link | safeHTML }} {{- end -}}
